func reviewMirror(t *testing.T) *Mirror {
 t.Helper()
 m, e := OpenMirror(filepath.Join(t.TempDir(), "messages.sqlite3"), "account-id", true)
 if e != nil { t.Fatal(e) }
 t.Cleanup(func(){m.Close()})

}
func reviewRemote(t *testing.T, f func(*http.Request, []any) any) *Remote {
 t.Helper()
 r := NewRemote(&Session{AccountID:"account-id",Cookies:map[string]string{"SAPISID":"synthetic"},PublicKey:"public-key"}, 0)
 r.HTTP.Transport = roundTripFunc(func(q *http.Request)(*http.Response,error){
  raw,_:=io.ReadAll(q.Body);var args []any;json.Unmarshal(raw,&args)
  v:=any([]any{[]any{"account-id"}})
  if q.URL.Path!="/voice/v1/voiceclient/account/get" {v=f(q,args)}
  b,_:=json.Marshal(v)
  return &http.Response{StatusCode:200,Header:http.Header{},Body:io.NopCloser(strings.NewReader(string(b)))},nil
 })
 return r
}
func TestReviewLatestRefreshGap(t *testing.T) {
 m:=reviewMirror(t);ctx:=context.Background()
 m.DB.Exec("INSERT INTO voice_runs(id,started_at,status) VALUES(1,'synthetic','bounded')")
 b,_:=json.Marshal([]any{threadFixture("t-1","")});p,_:=DecodePage(b,true)
 if _,e:=m.commit(ctx,1,"refresh","t-1","","",p.Threads);e!=nil{t.Fatal(e)}
 latest:=threadFixture("t-1","gap-cursor")
 latest[2].([]any)[0].([]any)[0]="newest"
 calls:=[]string{}
 r:=reviewRemote(t,func(q *http.Request,args []any)any{
  if strings.HasSuffix(q.URL.Path,"/list"){return []any{[]any{latest},""}}
  calls=append(calls,str(args[2]));return []any{latest}
 })
 out,e:=m.Sync(ctx,r,[]string{"texts"},5);if e!=nil{t.Fatal(e)}
 var cursor string;var complete bool
 m.DB.QueryRow("SELECT older_cursor,history_complete FROM voice_threads WHERE id='t-1'").Scan(&cursor,&complete)
 t.Logf("result=%+v get cursors=%q saved cursor=%q complete=%v",out,calls,cursor,complete)
 if complete||cursor==""||out.Status=="observed_exhausted" {t.Error("new latest-page continuation was discarded and a missing refresh interval is reported exhausted")}
}
