func TestRound2GapWhileDeepHistoryPending(t *testing.T){
 m,e:=OpenMirror(filepath.Join(t.TempDir(),"mirror.sqlite3"),"account-id",true);if e!=nil{t.Fatal(e)};defer m.Close()
 ctx:=context.Background();m.DB.Exec("INSERT INTO voice_runs(id,started_at,status) VALUES(1,'synthetic','bounded')")
 old:=Thread{ID:"t",Messages:[]Message{r2Message("old-anchor",1700000000000)},Contacts:[]Contact{{Phone:"+15555550100"}},OlderCursor:"deep"}
 if _,e=m.commit(ctx,1,"refresh","t","","deep",[]Thread{old});e!=nil{t.Fatal(e)}
 fresh:=old;fresh.Messages=[]Message{r2Message("new",1700001000000)};fresh.OlderCursor="gap"
 if _,e=m.commit(ctx,1,"refresh","t","","gap",[]Thread{fresh});e!=nil{t.Fatal(e)}
 // An inventory refresh arrives while the refresh gap is pending.
 if _,e=m.commit(ctx,1,"inventory","texts","","",[]Thread{fresh});e!=nil{t.Fatal(e)}
 bridge:=old;bridge.OlderCursor="still-has-old";bridge.Messages=[]Message{r2Message("middle",1700000500000),old.Messages[0]}
 if _,e=m.commit(ctx,1,"refresh_gap","t","gap",bridge.OlderCursor,[]Thread{bridge});e!=nil{t.Fatal(e)}
 var gap,older string;var dirty bool;m.DB.QueryRow("SELECT refresh_cursor,older_cursor,needs_refresh FROM voice_threads WHERE id='t'").Scan(&gap,&older,&dirty)
 if gap!=""||older!="deep"||!dirty{t.Fatalf("gap/deep/refresh axes conflated: %q %q %v",gap,older,dirty)}
}
